cmake_minimum_required(VERSION 3.16)
project(stock_px4_supervisor)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pixhawk_interfaces REQUIRED)
add_executable(stock_px4_supervisor src/stock_px4_supervisor.cpp)
add_executable(differential_advisor src/differential_advisor.cpp)
target_include_directories(differential_advisor PRIVATE include)
ament_target_dependencies(differential_advisor rclcpp std_msgs pixhawk_interfaces)
target_include_directories(stock_px4_supervisor PRIVATE include)
ament_target_dependencies(stock_px4_supervisor rclcpp sensor_msgs std_msgs pixhawk_interfaces)
install(TARGETS stock_px4_supervisor differential_advisor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(stock_px4_safety_tests test/test_safety.cpp)
  target_include_directories(stock_px4_safety_tests PRIVATE include)
  add_test(NAME stock_px4_safety_tests COMMAND stock_px4_safety_tests)
  add_executable(differential_kinematics_tests test/test_differential.cpp)
  target_include_directories(differential_kinematics_tests PRIVATE include)
  add_test(NAME differential_kinematics_tests COMMAND differential_kinematics_tests)
endif()
ament_package()
