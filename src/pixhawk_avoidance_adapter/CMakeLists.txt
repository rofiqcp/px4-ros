cmake_minimum_required(VERSION 3.16)
project(pixhawk_avoidance_adapter)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pixhawk_interfaces REQUIRED)
add_executable(obstacle_advisory_node src/obstacle_advisory_node.cpp)
ament_target_dependencies(obstacle_advisory_node rclcpp sensor_msgs geometry_msgs std_msgs tf2_ros pixhawk_interfaces)
install(TARGETS obstacle_advisory_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
